La machine à états
Gérer les écrans du jeu avec gameState

Un jeu complet n'est pas qu'un canvas avec un personnage — c'est une série d'écrans que le joueur traverse : titre, menu, jeu, pause, fin… Chaque écran est un état.

Principe : Une variable gameState stocke l'état actuel. La boucle principale regarde cet état et affiche le bon écran. Pour changer d'écran, il suffit de changer la variable.

// Toutes les valeurs possibles de gameState
const ETATS = {
  TITRE:    'titre',
  MENU:     'menu',
  NIVEAUX:  'niveaux',
  JEU:      'jeu',
  PAUSE:    'pause',
  VICTOIRE: 'victoire',
  GAME_OVER:'gameover',
  SCORES:   'scores'
};

let gameState = ETATS.TITRE;

// La boucle principale délègue à l'écran actif
function loop() {
  ctx.clearRect(0, 0, W, H);
  switch (gameState) {
    case ETATS.TITRE:    drawTitre();    break;
    case ETATS.MENU:     drawMenu();     break;
    case ETATS.NIVEAUX:  drawNiveaux();  break;
    case ETATS.JEU:      updateJeu(); drawJeu(); break;
    case ETATS.PAUSE:    drawPause();    break;
    case ETATS.VICTOIRE: drawVictoire(); break;
    case ETATS.GAME_OVER:drawGameOver(); break;
    case ETATS.SCORES:   drawScores();   break;
  }
  requestAnimationFrame(loop);
}

// Changer d'état — aussi simple que ça !
function allerAuMenu() { gameState = ETATS.MENU; }

2
Écran titre & Menu principal
Dessiner des boutons cliquables sur le canvas

L'écran titre accueille le joueur. Le menu propose les options. Les deux sont dessinés sur le canvas — on détecte les clics avec addEventListener('click').

// Structure d'un bouton canvas
const boutons = [
  { label: '▶ Jouer',     action: () => { gameState = ETATS.NIVEAUX; } },
  { label: '🏆 Scores',   action: () => { gameState = ETATS.SCORES;  } },
  { label: '⚙ Options',  action: () => { gameState = ETATS.OPTIONS; } }
];

function drawMenu() {
  // Fond dégradé
  const grad = ctx.createLinearGradient(0, 0, 0, H);
  grad.addColorStop(0, '#1A0810');
  grad.addColorStop(1, '#3D0020');
  ctx.fillStyle = grad;
  ctx.fillRect(0, 0, W, H);

  boutons.forEach(function(btn, i) {
    const bx = W/2 - 120, by = 80 + i * 60;
    const bw = 240, bh = 46;

    // Rectangle arrondi
    ctx.fillStyle = btn.hovered ? '#F43F5E' : '#2D0010';
    ctx.strokeStyle = '#F43F5E';
    ctx.lineWidth = 1.5;
    roundRect(ctx, bx, by, bw, bh, 12);
    ctx.fill(); ctx.stroke();

    // Texte
    ctx.fillStyle = btn.hovered ? '#fff' : '#FB7185';
    ctx.font = '700 18px Nunito';
    ctx.textAlign = 'center';
    ctx.fillText(btn.label, bx + bw/2, by + bh/2 + 6);
    btn.rect = { x: bx, y: by, w: bw, h: bh };
  });
}

// Détecter le clic sur les boutons
canvas.addEventListener('click', function(e) {
  const r = canvas.getBoundingClientRect();
  const mx = e.clientX - r.left, my = e.clientY - r.top;
  boutons.forEach(function(btn) {
    if (btn.rect && mx >= btn.rect.x && mx <= btn.rect.x+btn.rect.w
       && my >= btn.rect.y && my <= btn.rect.y+btn.rect.h) {
      btn.action();
    }
  });
});
💡 Effet hover : Écoute mousemove sur le canvas pour détecter si la souris survole un bouton, et mets btn.hovered = true/false. Le prochain rendu affiche la version surlignée.

Sélecteur de niveaux
Cartes de niveaux, verrouillage et progression

Le sélecteur affiche les niveaux disponibles sous forme de cartes. Les niveaux non encore débloqués apparaissent grisés avec un cadenas.

// Données des niveaux
const NIVEAUX = [
  { id:1, nom:'La forêt',    couleur:'#16A34A', requis:0  },
  { id:2, nom:'Les grottes', couleur:'#0284C7', requis:1  },
  { id:3, nom:'Le volcan',  couleur:'#DC2626', requis:2  },
  { id:4, nom:'???',        couleur:'#6D28D9', requis:3  }
];

let niveauxDebloqués = 1;  // chargé depuis localStorage

function drawNiveaux() {
  NIVEAUX.forEach(function(niv, i) {
    const col = i % 2, row = Math.floor(i / 2);
    const cx = 60 + col * 240, cy = 80 + row * 130;
    const debloque = niveauxDebloqués >= niv.requis + 1;

    // Carte du niveau
    ctx.fillStyle = debloque ? niv.couleur : '#444';
    roundRect(ctx, cx, cy, 200, 100, 14); ctx.fill();

    // Nom ou cadenas
    ctx.fillStyle = '#fff';
    ctx.font = '800 16px Nunito'; ctx.textAlign = 'center';
    ctx.fillText(debloque ? niv.nom : '🔒', cx+100, cy+50);
    if (debloque) {
      ctx.font = '600 12px Nunito';
      ctx.fillText('Niveau ' + niv.id, cx+100, cy+70);
    }
  });
}

// Au clic : lancer le niveau sélectionné
function lancerNiveau(n) {
  niveauActuel = n;
  gameState = ETATS.JEU;
  chargerNiveau(n);
}

Tableau de scores
localStorage pour la persistance des données

localStorage permet de sauvegarder des données dans le navigateur — elles persistent même après avoir fermé l'onglet.

localStorage ne stocke que des chaînes de caractères. Pour stocker un tableau ou un objet, utilise JSON.stringify() pour sauvegarder et JSON.parse() pour recharger. 
// Charger les scores (max 10 entrées)
function chargerScores() {
  const raw = localStorage.getItem('monJeu_scores');
  return raw ? JSON.parse(raw) : [];
}

// Sauvegarder un nouveau score
function sauvegarderScore(nom, points, niveau) {
  const scores = chargerScores();
  scores.push({ nom, points, niveau, date: new Date().toLocaleDateString() });
  scores.sort((a, b) => b.points - a.points);  // meilleurs en premier
  const top10 = scores.slice(0, 10);           // garder seulement le top 10
  localStorage.setItem('monJeu_scores', JSON.stringify(top10));
}

// Effacer tous les scores
function effacerScores() {
  localStorage.removeItem('monJeu_scores');
}

// Sauvegarder la progression (niveaux débloqués)
function sauvegarderProgression() {
  localStorage.setItem('monJeu_niveaux', niveauxDebloqués.toString());
}
function chargerProgression() {
  niveauxDebloqués = parseInt(localStorage.getItem('monJeu_niveaux') || '1');
}

Structure d'un niveau
Données, chargement et transitions

Chaque niveau est un objet de données : plateformes, ennemis, position de départ, objectif, temps imparti. On charge ces données quand on lance le niveau.

// Définition d'un niveau
const NIVEAU_DATA = [
  {
    id: 1,
    fond: '#0D2B1A',
    tempsMax: 60,         // secondes
    nbPieces: 8,
    posDepart: { x: 60, y: 300 },
    plateformes: [
      { x:0,   y:340, w:600, h:20 },   // sol
      { x:100, y:260, w:120, h:14 },
      { x:300, y:200, w:100, h:14 }
    ],
    ennemis: [
      { x:250, y:320, patrol: [200, 350] }
    ]
  },
  {
    id: 2, tempsMax: 45, nbPieces: 12,
    // ...
  }
];

let niveau = null;

function chargerNiveau(n) {
  niveau = JSON.parse(JSON.stringify(NIVEAU_DATA[n-1]));  // copie profonde
  joueur.x = niveau.posDepart.x;
  joueur.y = niveau.posDepart.y;
  pieces = genererPieces(niveau);
  timer = niveau.tempsMax * 60;  // en frames
}

// Fin de niveau
function niveauTerminé() {
  const score = calcScore(timer, niveau.nbPieces);
  sauvegarderScore(nomJoueur, score, niveau.id);
  niveauxDebloqués = Math.max(niveauxDebloqués, niveau.id + 1);
  sauvegarderProgression();
  gameState = ETATS.VICTOIRE;
}
💡 Copie profonde : JSON.parse(JSON.stringify(data)) copie l'objet entièrement. Indispensable pour ne pas modifier les données originales quand le joueur relance un niveau.

Timer, Game Over & Victoire
Boucle de fin et transitions

Un temps imparti rend le jeu plus tendu. On décrémente un compteur à chaque frame, et on déclenche le Game Over quand il tombe à zéro.

let timer = 0;          // en frames (60fps → 60 frames/sec)
let nomJoueur = '';

function updateJeu() {
  // Décompte
  timer--;
  if (timer <= 0) {
    gameState = ETATS.GAME_OVER;
    return;
  }

  // Victoire : toutes les pièces ramassées
  if (pieces.every(p => p.ramassee)) {
    niveauTerminé();
    return;
  }

  // Mort du joueur
  if (joueur.y > H + 100) {
    gameState = ETATS.GAME_OVER;
    return;
  }

  updateJoueur();
  updateEnnemis();
}

function drawHUD() {
  const secondes = Math.ceil(timer / 60);
  ctx.fillStyle = secondes <= 10 ? '#F43F5E' : '#fff';
  ctx.font = '800 16px Nunito';
  ctx.textAlign = 'right';
  ctx.fillText('⏱ ' + secondes + 's', W - 16, 30);

  ctx.fillStyle = '#FCD34D';
  ctx.textAlign = 'left';
  ctx.fillText('🪙 ' + piecesRamassées + '/' + niveau.nbPieces, 16, 30);
}

function calcScore(timerRestant, nbPieces) {
  return Math.round(nbPieces * 1000 + (timerRestant / 60) * 100);
}
